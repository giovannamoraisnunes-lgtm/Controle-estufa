        function evaluateAutomationRules() {
            const s = appState.sensors;
            const sp = appState.setpoints;

            // Rule 1: Temperature Control (Exhaust vs Heater)
            if (s.temp > sp.tempMax) {
                appState.actuators.fan = true;
                appState.actuators.heater = false;
            } else if (s.temp < sp.tempMin) {
                appState.actuators.heater = true;
                appState.actuators.fan = false;
            } else {
                appState.actuators.fan = false;
                appState.actuators.heater = false;
            }

            // Rule 2: Air Humidity Control (Fogger vs Fan)
            if (s.humidity < sp.humMin) {
                appState.actuators.fogger = true;
            } else if (s.humidity > sp.humMax) {
                appState.actuators.fogger = false;
                appState.actuators.fan = true; // Extra exhaust needed for humidity
            } else if (s.temp <= sp.tempMax) {
                appState.actuators.fogger = false;
            }

            // Rule 3: Soil Irrigation Pump Control
            if (s.soil < sp.soilMin) {
                appState.actuators.pump = true;
            } else if (s.soil >= sp.soilMax) {
                appState.actuators.pump = false;
            }

            // Rule 4: Grow Light Supplementation
            if (s.light < sp.lightMin) {
                appState.actuators.light = true;
            } else {
                appState.actuators.light = false;
            }
        }

        async function toggleWebSerial() {
            if (!('serial' in navigator)) {
                showToast('Navegador incompatível com Web Serial API! Use Google Chrome ou MS Edge.', 'error');
                return;
            }

            if (appState.serialPort) {
                // Close Connection
                try {
                    if (appState.serialReader) {
                        await appState.serialReader.cancel();
                    }
                    await appState.serialPort.close();
                    appState.serialPort = null;
                    updateSerialStatus(false);
                    showToast('Desconectado da Porta Serial USB.', 'info');
                } catch (e) {
                    showToast('Erro ao fechar porta: ' + e.message, 'error');
                }
                return;
            }

            try {
                // Request Port Selection
                const baud = parseInt(document.getElementById('serialBaudRate').value);
                appState.serialPort = await navigator.serial.requestPort();
                await appState.serialPort.open({ baudRate: baud });
                
                updateSerialStatus(true);
                showToast('Conectado à Porta Serial USB!', 'success');
                readSerialData();
            } catch (err) {
                console.error(err);
                showToast('Falha na conexão Serial: ' + err.message, 'error');
            }
        }

        async function readSerialData() {
            const textDecoder = new TextDecoderStream();
            const readableStreamClosed = appState.serialPort.readable.pipeTo(textDecoder.writable);
            const reader = textDecoder.readable.getReader();
            appState.serialReader = reader;

            let buffer = "";

            try {
                while (true) {
                    const { value, done } = await reader.read();
                    if (done) break;
                    if (value) {
                        buffer += value;
                        const lines = buffer.split('\n');
                        buffer = lines.pop(); // Keep unfinished tail

                        for (const line of lines) {
                            if (line.trim()) {
                                parseSerialJson(line.trim());
                            }
                        }
                    }
                }
            } catch (error) {
                console.error('Erro de Leitura Serial:', error);
            } finally {
                reader.releaseLock();
            }
        }

        function parseSerialJson(jsonString) {
            const consoleEl = document.getElementById('serialConsole');
            const time = new Date().toLocaleTimeString();
            
            // Append raw string to console
            const lineDiv = document.createElement('div');
            lineDiv.innerHTML = `<span class="text-gray-500">[${time}] RX:</span> ${jsonString}`;
            consoleEl.appendChild(lineDiv);
            consoleEl.scrollTop = consoleEl.scrollHeight;

            try {
                const data = JSON.parse(jsonString);
                if (data.temp !== undefined) appState.sensors.temp = parseFloat(data.temp);
                if (data.hum !== undefined) appState.sensors.humidity = parseFloat(data.hum);
                if (data.soil !== undefined) appState.sensors.soil = parseFloat(data.soil);
                if (data.light !== undefined) appState.sensors.light = parseFloat(data.light);
                
                // Disable Simulator automatically when real hardware feeds data
                if (appState.simulationActive) {
                    toggleSimulationMode(false);
                    document.getElementById('simToggle').checked = false;
                }
            } catch (e) {
                // Not JSON string, normal print
            }
        }

        function updateCppCode() {
            const ssid = document.getElementById('cfgWifiSsid').value || 'SuaRede';
            const pass = document.getElementById('cfgWifiPass').value || 'SuaSenha';
            const pinDht = document.getElementById('pinDht').value || '4';
            const pinSoil = document.getElementById('pinSoil').value || '34';
            const pinLdr = document.getElementById('pinLdr').value || '35';
            const pinPump = document.getElementById('pinRelePump').value || '26';

            const cppCode = `/*
 * EcoFlora IoT - Firmware para ESP32
 * Gerado Automaticamente pelo Painel Web
 */

#include <WiFi.h>
#include <DHT.h>

// Configurações da Rede Wi-Fi
const char* WIFI_SSID = "${ssid}";
const char* WIFI_PASS = "${pass}";

// Definição de Pinos dos Sensores e Relés
#define DHTPIN ${pinDht}
#define DHTTYPE DHT22
#define PIN_SOIL ${pinSoil}
#define PIN_LDR ${pinLdr}
#define PIN_RELE_PUMP ${pinPump}
#define PIN_RELE_FAN 27
#define PIN_RELE_LIGHT 14
#define PIN_RELE_FOGGER 12

DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(115200);
  dht.begin();
  
  pinMode(PIN_RELE_PUMP, OUTPUT);
  pinMode(PIN_RELE_FAN, OUTPUT);
  pinMode(PIN_RELE_LIGHT, OUTPUT);
  pinMode(PIN_RELE_FOGGER, OUTPUT);
  
  // Desliga todos os relés por padrão (Lógica invertida comum)
  digitalWrite(PIN_RELE_PUMP, HIGH);
  digitalWrite(PIN_RELE_FAN, HIGH);
  digitalWrite(PIN_RELE_LIGHT, HIGH);
  digitalWrite(PIN_RELE_FOGGER, HIGH);

  WiFi.begin(WIFI_SSID, WIFI_PASS);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
}

void loop() {
  // Leitura dos Sensores
  float t = dht.readTemperature();
  float h = dht.readHumidity();
  int rawSoil = analogRead(PIN_SOIL);
  int rawLdr = analogRead(PIN_LDR);

  // Mapeamento percentual dos sensores analógicos
  float soilPct = map(rawSoil, 3500, 1400, 0, 100);
  soilPct = constrain(soilPct, 0, 100);
  
  float lightLux = map(rawLdr, 0, 4095, 0, 15000);

  // Formatação e Envio de String JSON via Serial USB
  if (!isnan(t) && !isnan(h)) {
    Serial.print("{\\"temp\\":"); Serial.print(t, 1);
    Serial.print(",\\"hum\\":"); Serial.print(h, 1);
    Serial.print(",\\"soil\\":"); Serial.print(soilPct, 1);
    Serial.print(",\\"light\\":"); Serial.print(lightLux, 0);
    Serial.println("}");
  }

  delay(2000); // Intervalo de telemetria
}`;

            document.getElementById('cppCodeDisplay').innerText = cppCode;
        }

        function updateSimValue(key, val) {
            val = parseFloat(val);
            if (key === 'temp') {
                appState.sensors.temp = val;
                document.getElementById('simValTemp').innerText = `${val.toFixed(1)}°C`;
            } else if (key === 'hum') {
                appState.sensors.humidity = val;
                document.getElementById('simValHum').innerText = `${val}%`;
            } else if (key === 'soil') {
                appState.sensors.soil = val;
                document.getElementById('simValSoil').innerText = `${val}%`;
            } else if (key === 'light') {
                appState.sensors.light = val;
                document.getElementById('simValLight').innerText = `${Math.round(val)} Lux`;
            }
        }

        function applySimPreset(preset) {
            if (preset === 'ideal') {
                setSimValues(24.0, 65, 60, 8000);
                showToast('Preset aplicado: Dia Ideal', 'info');
            } else if (preset === 'hot_dry') {
                setSimValues(34.5, 32, 22, 12500);
                showToast('Preset aplicado: Calor e Seca Extrema', 'warning');
            } else if (preset === 'cold_night') {
                setSimValues(14.0, 75, 55, 200);
                showToast('Preset aplicado: Noite Fria', 'info');
            } else if (preset === 'storm') {
                setSimValues(21.0, 92, 88, 1200);
                showToast('Preset aplicado: Tempestade', 'info');
            }
        }

        function setSimValues(t, h, s, l) {
            document.getElementById('sliderSimTemp').value = t;
            document.getElementById('sliderSimHum').value = h;
            document.getElementById('sliderSimSoil').value = s;
            document.getElementById('sliderSimLight').value = l;

            updateSimValue('temp', t);
            updateSimValue('hum', h);
            updateSimValue('soil', s);
            updateSimValue('light', l);
        }

        function simulateDrift() {
            // Apply slight ambient thermal & humidity physics based on active actuators
            if (appState.actuators.heater) appState.sensors.temp += 0.2;
            if (appState.actuators.fan) {
                appState.sensors.temp -= 0.15;
                appState.sensors.humidity -= 0.3;
            }
            if (appState.actuators.fogger) appState.sensors.humidity += 0.4;
            if (appState.actuators.pump) appState.sensors.soil += 0.5;

            // Natural small variation noise (-0.05 to +0.05)
            appState.sensors.temp += (Math.random() - 0.5) * 0.1;
            appState.sensors.humidity += (Math.random() - 0.5) * 0.2;
            appState.sensors.soil -= 0.02; // Natural evaporation

            // Clamp bounds
            appState.sensors.temp = Math.max(5, Math.min(50, appState.sensors.temp));
            appState.sensors.humidity = Math.max(10, Math.min(100, appState.sensors.humidity));
            appState.sensors.soil = Math.max(0, Math.min(100, appState.sensors.soil));
        }

        function addLogEntry(source, message) {
            const timestamp = new Date().toLocaleTimeString();
            const activeList = Object.keys(appState.actuators)
                .filter(k => appState.actuators[k])
                .map(k => k.toUpperCase())
                .join(', ') || 'Nenhum';

            const entry = {
                timestamp,
                source,
                message,
                temp: appState.sensors.temp.toFixed(1),
                hum: appState.sensors.humidity.toFixed(1),
                soil: appState.sensors.soil.toFixed(1),
                light: Math.round(appState.sensors.light),
                actuators: activeList
            };

            appState.logs.unshift(entry);
            if (appState.logs.length > 50) appState.logs.pop();

            // Render Row in Table
            const tbody = document.getElementById('logsTableBody');
            if (tbody) {
                const tr = document.createElement('tr');
                tr.className = "hover:bg-darkcard/50 transition-colors";
                tr.innerHTML = `
                    <td class="p-3 text-gray-400">${entry.timestamp}</td>
                    <td class="p-3 font-semibold text-brand-400">${entry.source}</td>
                    <td class="p-3 text-orange-400">${entry.temp}°C</td>
                    <td class="p-3 text-cyan-400">${entry.hum}%</td>
                    <td class="p-3 text-emerald-400">${entry.soil}%</td>
                    <td class="p-3 text-yellow-400">${entry.light} Lux</td>
                    <td class="p-3 text-gray-300 text-[11px]">${entry.actuators}</td>
                `;
                tbody.insertBefore(tr, tbody.firstChild);
            }
        }

        function exportLogsCSV() {
            if (appState.logs.length === 0) {
                showToast('Não há registros de logs para exportar.', 'warning');
                return;
            }

            let csvContent = "data:text/csv;charset=utf-8,Horario,Origem,Temperatura,Umidade_Ar,Umidade_Solo,Luminosidade,Atuadores\n";
            appState.logs.forEach(l => {
                csvContent += `${l.timestamp},${l.source},${l.temp},${l.hum},${l.soil},${l.light},"${l.actuators}"\n`;
            });

            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", `ecoflora_logs_${new Date().toISOString().slice(0,10)}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);

            showToast('Arquivo CSV baixado com sucesso!', 'success');
        }

        function showToast(message, type = 'info') {
            const container = document.getElementById('toastContainer');
            if (!container) return;

            const toast = document.createElement('div');
            
            let bgClass = "bg-gray-900 border-gray-700 text-white";
            let iconClass = "fa-circle-info text-blue-400";

            if (type === 'success') {
                bgClass = "bg-emerald-950 border-emerald-800 text-emerald-200";
                iconClass = "fa-circle-check text-emerald-400";
            } else if (type === 'warning') {
                bgClass = "bg-amber-950 border-amber-800 text-amber-200";
                iconClass = "fa-triangle-exclamation text-amber-400";
            } else if (type === 'error') {
                bgClass = "bg-rose-950 border-rose-800 text-rose-200";
                iconClass = "fa-circle-xmark text-rose-400";
            }

            toast.className = `p-3.5 rounded-xl border ${bgClass} shadow-xl flex items-center gap-3 text-xs pointer-events-auto transform transition-all duration-300 translate-y-2 opacity-0`;
            toast.innerHTML = `
                <i class="fa-solid ${iconClass} text-base"></i>
                <span class="flex-grow font-medium">${message}</span>
                <button onclick="this.parentElement.remove()" class="text-gray-400 hover:text-white"><i class="fa-solid fa-xmark"></i></button>
            `;

            container.appendChild(toast);

            // Animate in
            setTimeout(() => {
                toast.classList.remove('translate-y-2', 'opacity-0');
            }, 10);

            // Auto dismiss
            setTimeout(() => {
                toast.classList.add('opacity-0');
                setTimeout(() => toast.remove(), 300);
            }, 4000);
        }
